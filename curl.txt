#### /

while true; do
  curl -s https://frases.virtualti.net/ | jq .version
  sleep 0.2
done


while true; do
  v=$(curl -s https://frases.virtualti.net/ | jq -r .version)
  echo "$v -- $(date '+%H:%M:%S')"
  sleep 0.2
done
